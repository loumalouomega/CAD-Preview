\usetikzlibrary{arrows.meta,calc,shapes.geometric}
\begin{document}
\begin{tikzpicture}[line width=1.3pt, line cap=round, line join=round, x=1mm,y=1mm]

  % button's theme-dependent background.
  \fill[gray!15, even odd rule] (-10,-8) rectangle (10,8) (-4,-4) rectangle (4,4);
  \draw[thick] (-10,-8) rectangle (10,8);
  \draw[thick] (-4,-4) rectangle (4,4);
  % Drive arrow: explicit filled triangle tip (a `>=Stealth` tip vanishes at
  % 1em toolbar size — see icons/README.md), shaft stopping above the tip.
  \draw[thick] (0,13) -- (0,8);
  \filldraw[thick] (0,4.5) -- (-2,8) -- (2,8) -- cycle;
\end{tikzpicture}
\end{document}
